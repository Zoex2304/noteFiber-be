## 3. Full Source Code with Explanations

### 3.1 Database Configuration (pkg/database/connection.go)
```go
package database

import (
    "context"
    "log"
    "github.com/jackc/pgx/v5/pgxpool"
)

// ConnectDB establishes and verifies a connection to PostgreSQL
func ConnectDB(connectionString string) *pgxpool.Pool {
    var err error
    db, err := pgxpool.New(context.Background(), connectionString)
    if err != nil {
        log.Panicf("Unable to connect to DB: %v", err)
    }

    err = db.Ping(context.Background())
    if err != nil {
        log.Panicf("Failed to ping: %v", err)
    }

    return db
}
```

### 3.2 Database Interface (pkg/database/queryer.go)
```go
package database

import (
    "context"
    "github.com/jackc/pgx/v5"
    "github.com/jackc/pgx/v5/pgconn"
)

// DatabaseQueryer defines the interface for database operations
type DatabaseQueryer interface {
    Exec(ctx context.Context, sql string, arguments ...any) (pgconn.CommandTag, error)
    Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)
    QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
}
```

## 4. Database Integration Details

### 4.1 Database Schema
```sql
-- Session Management
CREATE TABLE chat_session (
    id UUID PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    deleted_at TIMESTAMP,
    is_deleted BOOLEAN DEFAULT FALSE
);

-- Message Storage
CREATE TABLE chat_message (
    id UUID PRIMARY KEY,
    chat TEXT NOT NULL,
    role VARCHAR(10) NOT NULL,
    chat_session_id UUID NOT NULL,
    created_at TIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    deleted_at TIMESTAMP,
    is_deleted BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (chat_session_id) REFERENCES chat_session(id)
);

-- Raw Message Storage (for training)
CREATE TABLE chat_message_raw (
    id UUID PRIMARY KEY,
    chat TEXT NOT NULL,
    role VARCHAR(10) NOT NULL,
    chat_session_id UUID NOT NULL,
    created_at TIMESTAMP NOT NULL,
    updated_at TIMESTAMP,
    deleted_at TIMESTAMP,
    is_deleted BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (chat_session_id) REFERENCES chat_session(id)
);
```
